---
title: "Add an agent to your app"
description: "Run Sessions from your server, stream their events to the browser, and answer tool calls."
---

Your server keeps the API key, creates a Session per conversation, and relays its events to the browser. When the agent calls a tool your app owns, the turn pauses until your server sends back the result.

## Create the Agent

Give the Agent a custom tool that your server will run:

```sh
curl --fail-with-body -X POST "$CHECKFU_BASE_URL/v1/agents" \
  -H "Authorization: Bearer $CHECKFU_API_KEY" -H "Content-Type: application/json" \
  -d '{
    "name": "Support",
    "model": "claude-sonnet-5-5",
    "system": "Help customers with their orders. Use lookup_order before you answer.",
    "tools": [{
      "type": "custom",
      "name": "lookup_order",
      "description": "Look up an order by ID and return its status.",
      "input_schema": { "type": "object", "properties": { "order_id": { "type": "string" } }, "required": ["order_id"] }
    }]
  }'
```

## The server

```ts tools.ts
import { createHash } from "node:crypto"
import type Checkfu from "@checkfu/sdk"
import type { JsonSessionEvent } from "@checkfu/sdk"

export async function relaySession(
	client: Pick<Checkfu, "sessions">,
	sessionId: string,
	after: number | undefined,
	signal: AbortSignal,
	write: (event: JsonSessionEvent) => void,
) {
	for await (const event of await client.sessions.events.stream(
		sessionId,
		after === undefined ? undefined : { after },
		{ signal },
	)) {
		write(event)
	}
}

type CustomCall = Extract<JsonSessionEvent, { type: "agent.custom_tool_use" }>
type Pause = Extract<JsonSessionEvent, { type: "session.status_idle" }>
export type ToolResult = {
	id: string
	type: "user.custom_tool_result"
	custom_tool_use_id: string
	content: [{ type: "text"; text: string }]
	is_error: boolean
}
type Results = {
	read: (key: string) => Promise<ToolResult | undefined>
	write: (key: string, result: ToolResult) => Promise<void>
}

export function createToolProcessor(
	client: Pick<Checkfu, "sessions">,
	tools: Record<string, (input: Record<string, unknown>) => Promise<unknown>>,
	results: Results,
) {
	const stillWaiting = async (sessionId: string, pause: Pause, callId: string) => {
		const session = await client.sessions.retrieve(sessionId)
		if (session.status !== "idle" || session.archived_at !== null) return false
		for await (const event of client.sessions.events.list(sessionId, {
			order: "desc",
			types: ["session.status_idle", "user.custom_tool_result"],
		})) {
			if (event.type === "user.custom_tool_result" && event.custom_tool_use_id === callId)
				return false
			if (event.type === "session.status_idle") return event.seq === pause.seq
		}
		return false
	}
	const findCall = async (sessionId: string, id: string) => {
		for await (const event of client.sessions.events.list(sessionId, {
			order: "desc",
			types: ["agent.custom_tool_use", "agent.mcp_tool_use"],
		})) {
			if (event.id === id) {
				return event.type === "agent.custom_tool_use" ? event : undefined
			}
		}
	}
	const answer = async (sessionId: string, pause: Pause, call: CustomCall) => {
		if (!(await stillWaiting(sessionId, pause, call.id))) return
		const key = createHash("sha256")
			.update(JSON.stringify([sessionId, call.id]))
			.digest("hex")
		let result = await results.read(key)
		if (result === undefined) {
			const evaluate = async () => {
				const run = Object.hasOwn(tools, call.name) ? tools[call.name] : undefined
				if (run === undefined) throw new Error("Unknown application tool")
				return JSON.stringify(await run(call.input)) ?? "null"
			}
			const [text, isError] = await evaluate().then(
				(out) => [out, false] as const,
				() => ["Application tool failed.", true] as const,
			)
			result = {
				id: "result-" + key,
				type: "user.custom_tool_result",
				custom_tool_use_id: call.id,
				content: [{ type: "text", text }],
				is_error: isError,
			}
			await results.write(key, result)
		}
		try {
			await client.sessions.events.send(sessionId, { events: [result] })
		} catch (error) {
			if (await stillWaiting(sessionId, pause, call.id)) throw error
		}
	}
	return async (sessionId: string, event: JsonSessionEvent) => {
		if (event.type !== "session.status_idle" || event.stop_reason.type !== "requires_action") return
		for (const id of event.stop_reason.event_ids) {
			const call = await findCall(sessionId, id)
			if (call !== undefined) await answer(sessionId, event, call)
		}
	}
}
```

```ts server.mts
import { createServer, type IncomingMessage, type ServerResponse } from "node:http"
import { text as readText } from "node:stream/consumers"
import { mkdir, readFile, rename, rm, writeFile } from "node:fs/promises"
import { join } from "node:path"
import { setTimeout as delay } from "node:timers/promises"
import Checkfu from "@checkfu/sdk"
import { createToolProcessor, relaySession, type ToolResult } from "./tools.ts"

const client = new Checkfu()
const resultDirectory = ".checkfu-tool-results"
const processEvent = createToolProcessor(
	client,
	{ lookup_order: async ({ order_id }) => ({ order_id, status: "shipped" }) },
	{
		read: async (key) => {
			try {
				return JSON.parse(await readFile(join(resultDirectory, key), "utf8")) as ToolResult
			} catch (error) {
				if ((error as NodeJS.ErrnoException).code === "ENOENT") return undefined
				throw error
			}
		},
		write: async (key, result) => {
			await mkdir(resultDirectory, { recursive: true, mode: 0o700 })
			const final = join(resultDirectory, key)
			const staged = final + "." + crypto.randomUUID()
			try {
				await writeFile(staged, JSON.stringify(result), { flag: "wx", mode: 0o600 })
				await rename(staged, final)
			} finally {
				await rm(staged, { force: true })
			}
		},
	},
)

const processing = new Set<string>()
function ownSession(sessionId: string) {
	if (processing.has(sessionId)) return
	processing.add(sessionId)
	const follow = async () => {
		let after: number | undefined
		for (;;) {
			try {
				for await (const event of await client.sessions.events.stream(sessionId, { after })) {
					await processEvent(sessionId, event)
					after = event.seq
					if (event.type === "session.status_terminated" || event.type === "session.archived")
						return
				}
			} catch (error) {
				if (error instanceof Checkfu.NotFoundError || error instanceof Checkfu.AuthenticationError)
					throw error
				console.error("Session tool processor will retry", sessionId)
			}
			await delay(1_000)
		}
	}
	void follow()
		.catch(() => console.error("Session tool processor stopped", sessionId))
		.finally(() => {
			processing.delete(sessionId)
		})
}

// Supply the still-active Session IDs restored by your application on restart.
for (const id of (process.env.ACTIVE_SESSION_IDS ?? "").split(",").filter(Boolean)) ownSession(id)

async function relay(req: IncomingMessage, res: ServerResponse, sessionId: string) {
	const last = Number(req.headers["last-event-id"])
	const controller = new AbortController()
	req.on("close", () => controller.abort())
	res.writeHead(200, { "content-type": "text/event-stream", "cache-control": "no-cache" })
	try {
		const after = Number.isInteger(last) && last >= 0 ? last : undefined
		await relaySession(client, sessionId, after, controller.signal, (event) => {
			res.write(`id: ${event.seq}\nevent: ${event.type}\ndata: ${JSON.stringify(event)}\n\n`)
		})
	} catch (error) {
		if (!controller.signal.aborted) console.error(error)
	}
	res.end()
}

const json = async (req: IncomingMessage) => JSON.parse((await readText(req)) || "{}")

createServer(async (req, res) => {
	const [, resource, sessionId, action] = new URL(
		req.url ?? "/",
		"http://localhost",
	).pathname.split("/")
	try {
		if (resource !== "sessions") {
			res.writeHead(404).end()
		} else if (req.method === "POST" && !sessionId) {
			const session = await client.waitForPending(
				await client.sessions.create({
					agent: process.env.AGENT_ID!,
					environment_id: process.env.ENVIRONMENT_ID!,
				}),
			)
			ownSession(session.id)
			res.end(JSON.stringify({ id: session.id }))
		} else if (req.method === "GET" && sessionId && action === "events") {
			await relay(req, res, sessionId) // In a real app, check that the user owns this Session.
		} else if (req.method === "POST" && sessionId && action === "messages") {
			const { text } = await json(req)
			await client.sessions.events.send(sessionId, {
				events: [{ type: "user.message", content: [{ type: "text", text }] }],
			})
			res.end()
		} else if (req.method === "POST" && sessionId && action === "confirmations") {
			const { tool_use_id, result } = await json(req)
			await client.sessions.events.send(sessionId, {
				events: [{ type: "user.tool_confirmation", tool_use_id, result }],
			})
			res.end()
		} else {
			res.writeHead(404).end()
		}
	} catch (error) {
		if (!res.headersSent)
			res.writeHead(error instanceof Checkfu.APIError ? error.status : 500).end(String(error))
	}
}).listen(3000, "127.0.0.1", () => console.log("Listening on http://127.0.0.1:3000"))
```

This local demo listens on loopback. For an application, authorize each session read, message, and confirmation against the signed-in user.

Start the server in one terminal with `CHECKFU_BASE_URL` and `CHECKFU_API_KEY` exported:

```sh
AGENT_ID=... ENVIRONMENT_ID=... npx tsx server.mts
```

Then drive it from a second terminal:

```sh
SESSION_ID=$(curl -s -X POST http://localhost:3000/sessions | jq -r .id)
curl -N "http://localhost:3000/sessions/$SESSION_ID/events" &
curl -X POST "http://localhost:3000/sessions/$SESSION_ID/messages" \
  -H "Content-Type: application/json" \
  -d '{"text":"Where is order 1042?"}'
```

You should see the tool call, the `requires_action` pause, your server's tool result, and the agent's final reply stream past.

## Stream events to the browser

Because the relay writes each event's `seq` as the SSE `id`, `EventSource` picks up where it left off after a reconnect while keeping the API key on your server:

```js
const source = new EventSource(`/sessions/${sessionId}/events`)
source.addEventListener("agent.message", (e) => showReply(JSON.parse(e.data)))
source.addEventListener("agent.mcp_tool_use", (e) => {
	const call = JSON.parse(e.data)
	if (call.evaluated_permission === "ask") askUser(call) // then POST /sessions/:id/confirmations
})
```

## Handle tool calls

A pause arrives as `session.status_idle` with stop reason `requires_action`. Answer every ID in `stop_reason.event_ids`; the turn resumes once the last answer arrives.

- Answer `agent.custom_tool_use` with `user.custom_tool_result`.
- Answer `agent.mcp_tool_use` that has `evaluated_permission: "ask"` with `user.tool_confirmation` (`result: "allow"` or `"deny"`).

The server owns one tool processor per Session, even with zero browsers connected. Browser streams only observe events. The processor checks that a replayed pause is still current and saves each result and its stable event ID before submitting it. A retry reuses the saved result. After a restart, supply the active Session IDs from your application and reuse the same private results directory. This example uses one server and a read-only lookup. For several servers, claim each call in a shared durable store. For a side-effecting tool, use the same call key with a downstream idempotency check or commit the application effect and result together. Reusing a Checkfu event ID only deduplicates the submitted answer. While paused, the Session waits without making any model requests. See [Tools](/concepts/tools) and [Resume a stream](/guides/resume-a-stream).
